Cloudflare DNS API Examples

JavaScript API Call

// Lấy DNS Records từ Cloudflare qua API
async function getCloudflareeDNS(domain) {
    try {
        const response = await fetch(`/index.php?action=cloudflare-dns&domain=${domain}&include_proxied=true&compare_dns=true`);
        const data = await response.json();
        
        if (data.success) {
            console.log('✅ DNS Records retrieved:', data.data);
            
            // Display statistics
            const stats = data.data.statistics;
            console.log(`📊 Statistics:
            - Total Records: ${stats.total_records}
            - Proxied: ${stats.proxied_records}  
            - DNS Only: ${stats.dns_only_records}`);
            
            // Display records by type
            const records = data.data.cloudflare_records;
            for (const [type, recordList] of Object.entries(records)) {
                console.log(`${type} Records:`, recordList);
            }
            
            return data.data;
        } else {
            console.error('❌ API Error:', data.error);
            return null;
        }
    } catch (error) {
        console.error('❌ Request failed:', error);
        return null;
    }
}

// Usage
getCloudflareeDNS('example.com').then(result => {
    if (result) {
        // Process DNS data
        displayDNSResults(result);
    }
});

Fetch với options nâng cao:

// Fetch với POST data
fetch('/index.php?action=cloudflare-dns', {
    method: 'POST',
    headers: {
        'Content-Type': 'application/x-www-form-urlencoded',
    },
    body: new URLSearchParams({
        'domain': 'yoursite.com',
        'include_proxied': 'true',
        'show_only_active': 'true',
        'compare_dns': 'true'
    })
})
.then(response => response.json())  
.then(data => {
    if (data.success) {
        // Success handling
        const cfRecords = data.data.cloudflare_records;
        const comparison = data.data.comparison;
        
        console.log('Cloudflare Records:', cfRecords);
        console.log('DNS Comparison:', comparison);
    }
});

cURL Command

# Lấy DNS Records cho domain
curl -X GET "http://localhost/Note/index.php?action=cloudflare-dns&domain=example.com&include_proxied=true&compare_dns=true" \
     -H "Accept: application/json"

POST Request với cURL:

# POST request với parameters
curl -X POST "http://localhost/Note/index.php?action=cloudflare-dns" \
     -H "Content-Type: application/x-www-form-urlencoded" \
     -d "domain=yoursite.com&include_proxied=true&show_only_active=true&compare_dns=true"

Parse JSON Response:

# Parse với jq
curl -s "http://localhost/Note/index.php?action=cloudflare-dns&domain=example.com" | jq '
.data.statistics
'

# Extract specific record types
curl -s "http://localhost/Note/index.php?action=cloudflare-dns&domain=example.com" | jq '
.data.cloudflare_records.A[]
'

PHP Integration

<?php
// Direct function call (nếu include CacheHandler.php)
require_once 'CacheHandler.php';

// Simulate POST data
$_POST['domain'] = 'example.com';
$_POST['include_proxied'] = 'true';
$_POST['compare_dns'] = 'true';

// Capture JSON output
ob_start();
getCloudflaresDNSRecords();
$jsonResponse = ob_get_clean();

$result = json_decode($jsonResponse, true);

if ($result && $result['success']) {
    $dnsData = $result['data'];
    
    echo "Domain: " . $dnsData['domain'] . "\n";
    echo "Zone ID: " . $dnsData['zone_id'] . "\n";
    echo "Total Records: " . $dnsData['statistics']['total_records'] . "\n";
    
    // Process different record types
    foreach ($dnsData['cloudflare_records'] as $type => $records) {
        echo "\n{$type} Records (" . count($records) . "):\n";
        foreach ($records as $record) {
            $proxy = $record['proxied'] ? '🟠' : '⚫';
            echo "  {$proxy} {$record['name']} → {$record['content']}\n";
        }
    }
} else {
    echo "Error: " . ($result['error'] ?? 'Unknown error');
}
?>

Sử dụng CloudflareAPI class trực tiếp:

<?php
require_once 'CloudflareAPI.php';
require_once 'config.php';

$cloudflare = new CloudflareAPI();

// Get zone for domain
$zones = $cloudflare->listZones(1, 20, 'example.com');
if (empty($zones['result'])) {
    die('Domain not found in Cloudflare account');
}

$zoneId = $zones['result'][0]['id'];

// Get DNS records
$records = $cloudflare->listDNSRecords($zoneId);

foreach ($records['result'] as $record) {
    echo sprintf("%s: %s → %s (%s)\n", 
        $record['type'],
        $record['name'], 
        $record['content'],
        $record['proxied'] ? 'Proxied' : 'DNS Only'
    );
}
?>
